macro(find_package)
    if("${ARGV0}" STREQUAL "ICU")
        _find_package(ICU 76 REQUIRED COMPONENTS data i18n uc)
    else()
        _find_package(${ARGV})
    endif()
endmacro()
function(ladybird_lib name short)
    set(generated)
    foreach(file ${GENERATED_SOURCES})
        list(APPEND generated "${CMAKE_CURRENT_BINARY_DIR}/${file}")
    endforeach()
    add_library(${name} STATIC ${SOURCES} ${generated})
    if(name STREQUAL "LibJS")
        target_compile_options(${name} PRIVATE -O1 -Wno-error=maybe-uninitialized)
    endif()
    if(NOT name STREQUAL "AK")
        target_link_libraries(${name} PRIVATE AK)
    endif()
    string(TOUPPER "${short}" upper)
    generate_export_header(${name} EXPORT_MACRO_NAME "${upper}_API"
        EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/Export.h")
endfunction()
function(invoke_py_generator name script input header cpp)
    cmake_parse_arguments(G "" "EXTRA_HEADER" "arguments" ${ARGN})
    set(h "${CMAKE_CURRENT_BINARY_DIR}/${header}")
    set(c "${CMAKE_CURRENT_BINARY_DIR}/${cpp}")
    get_filename_component(dir "${h}" DIRECTORY)
    file(MAKE_DIRECTORY "${dir}")
    set(extra)
    if(G_EXTRA_HEADER)
        list(APPEND extra -x "${CMAKE_CURRENT_BINARY_DIR}/${G_EXTRA_HEADER}")
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${LADYBIRD_SOURCE}/Meta/${script}"
        -h "${h}" -c "${c}" ${extra} ${G_arguments}
        COMMAND_ERROR_IS_FATAL ANY)
endfunction()
